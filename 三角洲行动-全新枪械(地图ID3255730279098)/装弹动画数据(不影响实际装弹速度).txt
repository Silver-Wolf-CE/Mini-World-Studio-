local RunService = game:GetService("RunService")

AnimControl={}
AnimControl.curAnim={}
AnimControl.EventPool={}
function AnimControl:Init()
	  RunService:BindToRenderStep("AnimUpdate", Enum.RenderPriority.Camera.Value+1, function( )
		     if next(AnimControl.curAnim) then
			         local ClipName=AnimControl.curAnim.ClipName
					 local AnimatorNode=AnimControl.curAnim.AnimatorNode
					 if AnimatorNode.Parent==nil then
					        AnimControl.curAnim={}
					 else
						 if AnimControl.EventPool[ClipName] then
								local PlayedTime=AnimatorNode:GetCurrentStatePlayedTime (0)
								local PlayedNum=math.floor(PlayedTime)
								if AnimControl.curAnim.PlayedNum~=PlayedNum then
								         for _,data in ipairs(AnimControl.EventPool[ClipName]) do
										        if data.state==false then
												       data.state=true
													   data.Func()
												end
										 end
										 AnimControl.curAnim.PlayedNum=PlayedNum
								end
								if AnimControl.EventPool[ClipName] then
										for _,data in ipairs(AnimControl.EventPool[ClipName]) do
											  if PlayedTime<(data.EventTime+PlayedNum) then
													  data.state=false
											  else
													  if data.state==false then
															 data.state=true
															 data.Func()
													  end
											  end
										end
								end
						 end
					 end
			 end
	  end)
end
