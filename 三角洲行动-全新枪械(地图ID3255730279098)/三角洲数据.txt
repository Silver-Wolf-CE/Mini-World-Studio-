-- 获取一个Id
function PlayerConsts.getRandomPlayerId()
	return PlayerConsts.randomPlayerIds[math.random(1, #PlayerConsts.randomPlayerIds)]
end

-- 获取头像[机器人随机,随后固定]
local gIconCacheMap = {}
function PlayerConsts.getHeadThumbImg(playerId , fmt)
	fmt = fmt or PlayerConsts.RobloxHeadThumb
	if playerId > 5000000000000 then
		local sKey = string.format("%d-%s" , playerId , fmt)
		if not gIconCacheMap[sKey] then
			local sImg = string.format(fmt, PlayerConsts.getRandomPlayerId())
			gIconCacheMap[sKey] = sImg
		end
		return gIconCacheMap[sKey]
	else
		return string.format(fmt , playerId)
	end
end
