local RunService = game:GetService("RunService")
local MGM = {}

local _static_client_map = {}
local _static_server_map = {}

if RunService:IsServer() then
	if Const.VersionMode ~= 1 then
		return
	end
	local function OnCLIENT_GM_COMMAND(userId, msgid, cmd)
		MGM:ServerTrigger(userId, cmd.name, cmd.args)
	end

	Bridge:RegisterClientMessageCallback(Proto.ClientMSGID.CLIENT_GM_COMMAND, OnCLIENT_GM_COMMAND)
end

MGM.ClientRegister = function(self, command, func)
	if Const.VersionMode ~= 1 then
		return
	end
	local evs = _static_client_map[command]
	if evs == nil then
		evs = {}
		_static_client_map[command] = evs
	end
	
	if evs[func] ~= nil and evs[func]._valid == true then
		error("events already register, gm command " .. command)
		return false
	end
	
	evs[func] = {
		_valid = true,
		_callback = func
	}
	return true;
end

MGM.ServerRegister = function(self, command, func)
	command = string.upper(command) --gm界面输入框存在自动大写的问题，此处为了适应问题
	local evs = _static_server_map[command]
	if evs == nil then
		evs = {}
		_static_server_map[command] = evs
	end
	
	if evs[func] ~= nil and evs[func]._valid == true then
		error("events already register, gm command " .. command)
		return false
	end
	
	evs[func] = {
		_valid = true,
		_callback = func
	}
	return true;
end

MGM.ClientTrigger = function(self, userId, name, args)
    LOG('temp',"GM name : " .. name .. ", args : " .. OUtils.table2str(args))
    local evs = _static_client_map[name]
    if evs ~= nil then
        for k, v in pairs(evs) do
            v._callback(userId, args)
        end
    end
end
