    loadModel = function(self)
        self.handle = SandboxNode.new('Model')--MainStorage.Assets.Weapon[self.config.gun_res].Handle:Clone()
        self.handle.LocalSyncFlag = Enum.NodeSyncLocalFlag.DISABLE
        local skinId
        if self.skinId and self.skinId ~= 0 then

            skinId = CMainGame.itemManager:GetGunSkinIdByMatId(self.cfg_id, Config.Data.gun[self.cfg_id].default_skin_mat) or 0
        end
        local skinCfg = Config.Data.shop_item[skinId]

            if self.resource_cfg.Mod_Special == '' then

            end

        else
            self.handle.ModelId = self.resource_cfg.Mod
        end

        self.handle.Name = self.config.gun_res

        local _Muzzle = SandboxNode.new('BindAttachment')
        _Muzzle.Name = 'Muzzle'
        _Muzzle.Parent = self.handle
        self.handle.Muzzle.LocalPosition = Vector3.new(0,0,0)
        self.handle.Muzzle.LocalEuler = Vector3.new(0,0,0)
        self.handle.Muzzle.LocalScale = Vector3.new(1,1,1)

        self.handle.EnablePhysics = false
        self.handle.Visible = false
        self.handle.CastShadow = (not self.isLocalPlayer)
        self.handle.CollideGroupID = Const.CollisionType.INESSENTIAL
        self.handle_load_handler = self.handle.LoadFinish:connect(function(suc)
			self:setSkin(self.handle)
            if self.handle_load_handler then
                self.handle_load_handler:disconnect()
                self.handle_load_handler = nil
            end

                end)
            end

        self.handle.Parent = self.node
        -- self.handle.Visible = false
        self:SetHandle()

    --@func 设置枪械模式参数 
    setMode = function(self, _mode)
        self.mode = _mode
        if self.mode_datas[_mode] then
            self.cur_mode_data = OUtils.table.DeepCopy(self.mode_datas[_mode])
        end
        if self.cur_mode_data then
            self.magazineCapacity = self:_fixSkinAttribute(self.cur_mode_data.magazine, ItemDefine.EnumSkinAttrType.eBulletCapacity) 
            self.magazineCapacity_carry = self:_fixSkinAttribute(self.cur_mode_data.magazine_carry, ItemDefine.EnumSkinAttrType.eBulletTakeCapacity) 

            --这里先这样特殊处理，后期在数据配表层面做一些调整再修改逻辑
            local ammoIndex = self.mode
            if self.config.special_action == Const.WeaponSpecialAttack.CHANGE_MODE then
            else
                ammoIndex = 1
            end
            self.numOfBulletsInGun = self.Ammos[ammoIndex]
            self.numOfBulletsExtra = self.AmmoExtras[ammoIndex]

            self.fireDelay = self.cur_mode_data.fire_delay or 100
            self.muzzleVelocity = self.cur_mode_data.muzzle_velocity or 100 --子弹出膛速度
            self.reloadTime = self.cur_mode_data.reload_time--换子弹时间
            self.selectiveMode = self.cur_mode_data.shooting_mode[1]
            self.selectiveNum = self.cur_mode_data.shooting_mode[2] or 2
            self.maxRange = self.cur_mode_data.max_range
            self.fireInterval = self.cur_mode_data.fire_rate --子弹间隔
            self.sight_on_time = self.cur_mode_data.weapon_scaleTime or 100--开镜时间
            self.valueChangeForAccuracy_x:forceValue(math.min(self:_fixSkinAttribute(self.cur_mode_data.spread_max[1], ItemDefine.EnumSkinAttrType.eScatteringRange),self.cur_mode_data.spread_radius[1]),true)
            self.valueChangeForAccuracy_y:forceValue(0.5*OUtils.toRadian(self.cur_mode_data.spread_range[1]),true)
            self.cur_mode_id = self.mode_data_id[_mode]
        else
            self.cur_mode_data = OUtils.table.DeepCopy(self.mode_datas[1])
            self.cur_mode_id = self.mode_data_id[1]
        end

        --提前计算好皮肤修正后的后坐力, 散射上限等数据
        for i = 1, 5 do 
            self.cur_mode_data.spread_max[i] = self:_fixSkinAttribute(self.cur_mode_data.spread_max[i], ItemDefine.EnumSkinAttrType.eScatteringRange)
            self.cur_mode_data.vertical_recoil_max[i] = self:_fixSkinAttribute(self.cur_mode_data.vertical_recoil_max[i], ItemDefine.EnumSkinAttrType.eRecoil)
            self.cur_mode_data.horizontal_recoil_max[i] = self:_fixSkinAttribute(self.cur_mode_data.horizontal_recoil_max[i], ItemDefine.EnumSkinAttrType.eRecoil)
        end

        if self.isLocalPlayer then
            --更新弹匣数据
            self.host:updateAmmo(self:getCapacity(), self:getCapacityCarry())
        end

        if self.cfg_id ~= 2825 then  --2825改成 SPECAIL_TIME 事件触发
            self:playEntityBulletEffect()
        end
    end,

    --[[
        修正枪械皮肤带来的属性加成 
        @base = 配置表中对应的值：弹匣/携弹最大容量/后坐力上限/散射范围 
        @extendType = ItemDefine.EnumSkinAttrType
	]]
    _fixSkinAttribute = function(self, base, extendType)
        if not base or base <= 0 then
            return base
        end
        --如果是携弹容量.策划说先判断是否购买了双倍弹药特权
        if extendType == ItemDefine.EnumSkinAttrType.eBulletTakeCapacity then
            local actInfo = CMainGame.accountManager:GetAccountInfo()
            if actInfo and actInfo.DoubleBulletPrivilege then
                base = base * 2
            end
        end
		--皮肤加成弹匣容量和携弹容量
        local extendAttr = CMainGame.equipManager:GetExtendAttrByType(self.player.UserId, extendType, self.cfg_id)
        if extendAttr then
            if extendAttr.valueType == ItemDefine.EnumValueType.eExplicit then
                --绝对值
                base = base + extendAttr.value
            else
                --百分比
                base = base * (1 + extendAttr.value)
                --携弹容量.向上取整
                if extendType == ItemDefine.EnumSkinAttrType.eBulletTakeCapacity then
                    base = math.ceil(base)
                end
            end
        end
		return base
	end,

    _getStateIndex = function(self)
        local stanceState = self.host:GetStanceState()
        local moveState = self.host:GetMoveState()
        local index = 1 --默认站立
        if self:_isSnipeAim() or self:_isMaunalAim() then
            index = 5 --开镜或机瞄下
        elseif stanceState == Const.CharactorStanceState.JUMP or stanceState == Const.CharactorStanceState.FLYING then
            index = 4
        elseif stanceState == Const.CharactorStanceState.CROUCH then
            index = 3
        elseif moveState == Const.CharactorMoveState.MOVE then
            index = 2
        end
        return index
    end,

    --@func 计算后座导致的摄像机变化
    --@shotsFired 连射数
    --@返回
    _calcKickBack = function(self,shotsFired)
        local index = self:_getStateIndex()

        local recoil_pitch = self.cur_mode_data.vertical_recoil[index]
        local recoil_pitch_add = self.cur_mode_data.vertical_recoil_correct[index]
        local recoil_pitch_limit = self.cur_mode_data.vertical_recoil_max[index] 

        local recoil_yaw = self.cur_mode_data.horizontal_recoil[index]
        local recoil_yaw_add = self.cur_mode_data.horizontal_recoil_correct[index]
        local recoil_yaw_limit = self.cur_mode_data.horizontal_recoil_max[index]

		local v_p = self.cur_mode_data.vertical_recoil_proportion + math.random()*(1-self.cur_mode_data.vertical_recoil_proportion)
		local h_p = self.cur_mode_data.horizontal_recoil_proportion + math.random()*(1-self.cur_mode_data.horizontal_recoil_proportion)
		-- LOG('wj',"recioil value".. v_p)
        local pitch = -math.min(v_p * (recoil_pitch+shotsFired*recoil_pitch_add),recoil_pitch_limit)
        local yaw = math.min(h_p * (recoil_yaw+shotsFired*recoil_yaw_add),recoil_yaw_limit)
        self.change_probability = self.change_probability + math.random(self.cur_mode_data.change_probability[1],self.cur_mode_data.change_probability[2])
        if self.change_probability >= 100 then
            self.yaw_dir = -self.yaw_dir
            self.change_probability = 0
        end
        yaw = yaw * self.yaw_dir

        --设置给camera
		GameCamera:onGunRecoil(yaw,pitch)
        -- LOG('wj',"recioil pitch="..pitch..'  yaw='..yaw)
        GameCamera.shake_cal:shake({way='shake_recoil',yaw=yaw*self.cur_mode_data.offset_distance[index],pitch=pitch*self.cur_mode_data.offset_distance[index],time=self.cur_mode_data.return_time*0.001})
    end,

    --@func 设置武器状态
    --@参数weapon_state 武器状态
    --@返回
    _setWeaponState = function(self,weapon_state) --设置武器状态
--LOG('temp','fire setWeaponState...........'.. weapon_state)
        local oldState = self.weaponState
        self.weaponState = weapon_state
        -- print('SetWeaponState = '..weapon_state)
        self:_AfterWeaponStateChange(oldState)
        if oldState == Const.WeaponState.FIRING 
        and self._autoFire_On then
            self._autoDetect_timer = tick()
            self._autoFire_On = false
            self.hasActiveAutoFire = false
            self:releaseTheTrigger(true)
        end
        
        if self.weaponState == Const.WeaponState.FIRING then
            local _ison = CMainGame.playerManager:GetRoleByUserId(self.player.UserId)
            _ison:SetAimAssist(true)
            -- CMainGame.AimAssist = true

            if old_state == Const.WeaponState.READY then
                self:_updateBulletRemainingInBore()
            end

            if self.bulletRemainingInBore then

            self.weaponStateTimer = tick() + self.sight_on_time
        elseif self.weaponState == Const.WeaponState.PREPARING then
            self.firedRoundsPerPull = 0
            self.weaponStateTimer = tick() + self.prepareTime
        elseif self.weaponState == Const.WeaponState.COCKING then
            self.firedRoundsPerPull = 0
            --local bOldNotBulletRemainingInBore = not self.bulletRemainingInBore
			-- if (bOldNotBulletRemainingInBore and self.numOfBulletsInGun > 0 and self.supportBulletInBore) and self:_isManualBolt() then

                    end
				end

                if self.boltSoundNode and  self.isLocalPlayer then
                    SoundManager:play(self.boltSoundNode)
                end

			end
        elseif self.weaponState == Const.WeaponState.COCKING_PREPARE then
            self.firedRoundsPerPull = 0
            self.weaponStateTimer = tick()--self.desc.loadEndTime
        elseif self.weaponState == Const.WeaponState.DECOCKING then
            self.firedRoundsPerPull = 0
            if old_state == Const.WeaponState.IDLE or old_state == Const.WeaponState.FIRING or old_state == Const.WeaponState.RELOADING then

            end

                self.host:equipTool(3)
            end
        elseif self.weaponState == Const.WeaponState.HOLDING then
            self.host:onHolding()
        elseif self.weaponState == Const.WeaponState.PRE_FIRE then
            self.host:onPreFire()
        end
    end,

    --击中了玩家
    hitPlayer = function(self,start_pos,hit)
        --todo  remoteevent hitPlayer
        --self.hitPlayer.fireServer(hit.hitActorId)
        if self.isLocalPlayer then
            --LOG('zn','hitplayer',hit.obj.Name)
            local hitPart = hit.obj:GetAttribute("HitPart")
            local hitPlayer = hit.obj.Parent.Parent.Parent
            local hitUserId = hitPlayer:GetAttribute("UserId") or hitPlayer.UserId
  
            --计算一下是否是背后攻击
            local isBehind = 0
            local attack_dir = self.player.Character.Position - hitPlayer.Position
            attack_dir:Normalize()
            attack_dir = attack_dir - Vector3.new(0, attack_dir.Y, 0)

            local enemy_dir = hitPlayer.Rotation:LookDir()
            enemy_dir:Normalize()
            enemy_dir = enemy_dir - Vector3.new(0, enemy_dir.Y, 0)

            local dot_result = attack_dir:dot(enemy_dir)
            LOG("WWW","dot_result",dot_result,math.acos(dot_result),math.acos(dot_result) * 180 /  3.141592653)
            if dot_result > 0.5 then
                isBehind = 1
            end

            local tbl = {weapon_id=self.cfg_id,
            damage_type = Const.DamageType.NORMAL,
            hitPart=hitPart,
            hitDist=hit.distance,
            pos_x = hit.position.x,
            pos_y = hit.position.y,
            pos_z = hit.position.z,
            start_x = self.handle.Muzzle.Position.x,
            start_y = self.handle.Muzzle.Position.y,
            start_z = self.handle.Muzzle.Position.z,
            normal_x = hit.normal.x,
            normal_y = hit.normal.y,
            normal_z = hit.normal.z,
            aim_shot= self:_isSnipeAim(),
            blind_shoot = self:blindShoot(),
            hitBehind = isBehind,
            hitActorID=hitUserId,
            mode_id = self.cur_mode_id}

            Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_HIT, tbl)

            self:onHitPlayer(tbl)
        end
    end,

                bulletCreator.startPos = bulletCreator.startPos + Vector3.new(self.aiOffsetX,0,self.aiOffsetZ)
            else
                local euler = self.player.Character.Euler
                local rotate = Quaternion.FromEuler(euler.x, self.player.node:GetAttribute("Head_Pitch"), euler.z)
                bulletCreator.dir = rotate:LookDir()
            end
            -- end
        end
        --创建子弹
        local cur_accuracy_x = math.max(0, self.valueChangeForAccuracy_x.cur)
        bulletCreator.accuracy_x = cur_accuracy_x--self.accuracy_x
        bulletCreator.accuracy_y = self.valueChangeForAccuracy_y.cur--self.accuracy_y
        bulletCreator.CollisionGroup = self.host.bulletCollisionGroup
        bulletCreator.cur_mode_id = self.cur_mode_id

        local num = 1
        if self.special_attack and self.config.special_action == Const.WeaponSpecialAttack.SHOOT_ALL then
            num = self:getCapacity()
        end
        for _ = 1, num do
            local bullet = RayBullet.new()
            bullet:calHit(bulletCreator)
        end
        
        self:playFireSoundAndEffect()
        
        local _interval = self:_getFireInterval()
        -- self.weaponStateTimer = self.weaponStateTimer + _interval
        self.fireStateTimer = tick() + self:_getFireInterval()
        
        local reduce_num = 1
        if self.special_attack and self.config.special_action == Const.WeaponSpecialAttack.SHOOT_ALL then
            reduce_num = self:getCapacity()
        end
        self:reduceCapacity(reduce_num)
        self:_setBulletsAttribute(false)
        self.firedRoundsPerPull = self.firedRoundsPerPull + reduce_num

        if not self:checkCapacityValid()  then
            self:_setWeaponState(Const.WeaponState.DECOCKING)
        elseif self:_isPassAllRoundsPerPull() then--打完本轮的所有子弹了
            self:_setWeaponState(Const.WeaponState.IDLE)
        elseif  not self.hasPulledTheTrigger then
			self:_setWeaponState(Const.WeaponState.IDLE)
        end

        self:_updateBulletRemainingInBore()
        if self:_isManualBolt() then
            self.bulletRemainingInBore = false
            self.haveFinishedCOCKING_WeaponIsManualBolt = false
        end

        --通知当前持有武器数据变化
        if self.isLocalPlayer then
            --todo 通知界面
			self.host:updateAmmo(self:getCapacity(), self:getCapacityCarry())
            Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_FIRE,{weapon_id = self.cfg_id})
        end
    end,

    --@func 模拟武器状态
    --@参数deltaTime 间隔事件
    update = function(self,deltaTime)
        local msec = tick()
        --延迟关掉开火声音
        -- if self.delay_stopFireSound then
        --     self.delay_stopFireSound = self.delay_stopFireSound - deltaTime
        --     if self.delay_stopFireSound <= 0 then
        --         self:_stopFireSound()
        --     end
        -- end

        self:_autoFireDetect(msec)
        
        local _ison = CMainGame.playerManager:GetRoleByUserId(self.player.UserId)
        if _ison:GetIsAimAssist() then
            if msec - self.aimAssist_timer > 100 then
                if self.weaponState ~= Const.WeaponState.FIRING then
                    -- CMainGame.AimAssist = false
                    _ison:SetAimAssist(false)
                end
            end
        end

        --精度处理
        if self.weaponState ~= Const.WeaponState.FIRING and self.weaponState ~= Const.WeaponState.FIRING_EX then
            if self.delayFire then
				self.delayFire = false
				--[[if self.shotsFired > self.desc.accuracyMaxShotNum then
					self.shotsFired = self.desc.accuracyMaxShotNum
                end]]
                local index = self:_getStateIndex()
				self.decreaseShotsFiredTime = msec + self.cur_mode_data.recoil_cooling_time[index] --让停止攻击之后立即开始减少连射数
			end
        end

        --最大连射数恢复
        if self.weaponState ~= Const.WeaponState.FIRING and self.weaponState ~= Const.WeaponState.FIRING_EX and self.weaponState ~= Const.WeaponState.RELOADING then
            if self.shotsFired > 0 then
				if self.decreaseShotsFiredTime < msec then
					--self.decreaseShotsFiredTime = msec + self.config.recoil_cooling_time[1]
                    self.decreaseShotsFiredTime = 0
					self.shotsFired = 0--self.shotsFired - 1 --改为立即清空连射数
                end
			end
        end

        --[[local bUpdateAccuracy = false
		if self.weaponState ~= Const.WeaponState.RELOADING then
			--做精度调节参数的回缩
			if self.shotsFiredAdjustParaForAccuracy > 0 then
				static float fTemp_Weapon_fDecreaseShotsInterval = float(cfg_Weapon_fDecreaseShotsInterval);
				self.hotsFiredAdjustParaForAccuracy -= fElapsedTime/fTemp_Weapon_fDecreaseShotsInterval;
				bUpdateAccuracy = TRUE;
            end
		end]]

		self:_updateAccuracy(false)

		--精度的平滑过度
		if true then
            --精度平滑过度
			if self.valueChangeForAccuracy_x.changing then
                self.valueChangeForAccuracy_x:update(deltaTime)
				if self.cur_mode_data.crossHair_mode == 1 then
					self.isAccDirty = true
				end
            end
			if self.valueChangeForAccuracy_y.changing then
                self.valueChangeForAccuracy_y:update(deltaTime)
				if self.cur_mode_data.crossHair_mode == 1 then
					self.isAccDirty = true
				end
            end

            local cur_accuracy_x = math.max(0, self.valueChangeForAccuracy_x.cur)
            self.sightRange = cur_accuracy_x * 25 -- 此值不知道应该是多少，需要调整
            if self.isAccDirty then
                --通知当前主角状态变化
                if self.isLocalPlayer then
                    --_G.client.uiBridge:setGameAttrDirty(uiconst.GameAttrDirty.Aim)
					self.host:updateAim(self.sightRange)
				--LOG('temp',"sightrange = ".. self.sightRange)
                end
                self.isAccDirty = false
            end
        end


        --状态变化
        if self.weaponState == Const.WeaponState.FIRING then
            if msec > self.fireStateTimer then
                self:_tryFire()
            end
        elseif self.weaponState == Const.WeaponState.FIRING_EX then
            self:_tryFire()
            self:reload()
        elseif self.weaponState == Const.WeaponState.IDLE then
            if msec > self.weaponStateTimer then
				--处理单发枪，如果打完一发之后，需要拉枪栓，不管是否按住了左键
				if self:needLoad() and self:_isPassAllRoundsPerPull() then
                    -- self.haveFinishedCOCKING_WeaponIsManualBolt = false
					self:_setWeaponState(Const.WeaponState.COCKING_PREPARE) --拉枪栓子弹上膛
                elseif self:checkCapacityValid() then
                    self:_setWeaponState(Const.WeaponState.READY)
                    --[[if self.hasPulledTheTrigger then --又开枪了，继续开，否则置为等待开火状态
                        self:_setWeaponState(Const.WeaponState.FIRING)
                    else
                         self:_setWeaponState(Const.WeaponState.READY)
                    end]]
                end
            end
        elseif self.weaponState == Const.WeaponState.PORTING then
            if msec > self.weaponStateTimer then
                --self._portDone(host_obj)
                self:_setWeaponState(Const.WeaponState.READY)
            end
        -- elseif self.weaponState == Const.WeaponState.PREPARING then
        --     if msec > self.weaponStateTimer then
        --         if self:_isManualBolt() and not self.haveFinishedCOCKING_WeaponIsManualBolt then
        --             self.host:onLoad()
        --             self:_setWeaponState(Const.WeaponState.COCKING_PREPARE)
        --         else
        --             self:_setWeaponState(Const.WeaponState.READY)
        --         end
		-- 	end
        elseif self.weaponState == Const.WeaponState.COCKING then
            if msec > self.weaponStateTimer then
                if self.haveFinishedCOCKING_WeaponIsManualBolt then
                    if self:checkCapacityValid() then
                        self:_setWeaponState(Const.WeaponState.READY)
                        --实现上弹后返回开镜状态
                    -- elseif self.numOfBulletsExtra[self.mode] > 0 and self.isLocalPlayer then--auto reload
                    elseif self:checkCapacityCarry() and self.isLocalPlayer then
                        self:reload()
                    end
                end
			end
        elseif self.weaponState == Const.WeaponState.COCKING_PREPARE then
            if msec > self.weaponStateTimer then
                if not self.is_sight_on or self.hasReleasedTheTrigger then
                    self:_setWeaponState(Const.WeaponState.COCKING)
                end
			end
        elseif self.weaponState == Const.WeaponState.RELOADING then

        elseif self.weaponState == Const.WeaponState.DECOCKING then
            if msec > self.weaponStateTimer then
                if self.isLocalPlayer then
                    --如果是逐弹上弹的，并且还有子弹在枪里，则不应该出现在空仓的逻辑里，调回上弹逻辑
                    if self:_isOneByOneReloadType() and self:checkCapacityValid() then
                        self:_setWeaponState(Const.WeaponState.RELOADING)
                    end

                    if not self.is_sight_on or self.hasReleasedTheTrigger then
                         --强制关镜
                        if self.is_sight_on then
                            self:portArms(false,true)
                        end
                        --自动换蛋夹
                        -- if self.numOfBulletsExtra[self.mode] > 0 then
                        if self:checkCapacityCarry() then
                            if true then
                                self:reload()
                            end
                        else --没子弹了，看看有没有其他子弹
                            self:_resetWeaponState(not self.is_sight_on)
                        end
                    end
                end
            end
        elseif self.weaponState == Const.WeaponState.READY then
            if msec > self.weaponStateTimer then
                if self.hasPulledTheTrigger then
                    self:pullTheTrigger()
                end
            end
        end

    end,

    --@func 自动射击检测
    _autoFireDetect = function(self, _msec)
        if not self.enable_control then return end
        local _ison = CMainGame.playerManager:GetRoleByUserId(self.player.UserId)
        -- if not CMainGame.AutoFire then return end
        -- if not self.isLocalPlayer then return end
        local data= CMainGame.SettingManager:getValue(Const.Set.OPERATE_MODE)
        if not _ison:GetIsAutoFire() or not self.isLocalPlayer or data~=3  then
            return 
        end

        if not self._autoDetect_timer then
            self._autoDetect_timer = _msec
        end

        if (not self._autoFire_On) and (self.weaponState == Const.WeaponState.READY) then --判定阶段
            if self.host.hasPlayerActiveTool then
                return
            end
    
            if (_msec - self._autoDetect_timer) <= 10 then return end
            self._autoDetect_timer = _msec
            local startPos = game.WorkSpace.CurrentCamera.Position
            local dir = GameCamera.node.Rotation:LookDir()
            local ret = WS:SweepSphere(Config.Const.AutoFire_Config.AutoFire_Range, startPos, dir, self.auto_fire_dist, false, self.host.bulletCollisionGroup)
            if ret.isHit then
                if ret.obj then
                    if ret.obj:GetAttribute("HitPart") then
                        --检测到玩家
                        self._autoFire_On = true
                        -- print("#########已命中，目标的网格体为：", ret.obj.CollideGroupID,'  Name='..ret.obj.Name)
                        LOG('AutoFire', 'Hit Player, Part='..ret.obj:GetAttribute("HitPart"))
                    else
                        if CUR_PLATFORM == Enum.EnumDeviceType.WIN then
                            if self._auto_fire_detect_name ~= ret.obj.Name then
                                self._auto_fire_detect_name = ret.obj.Name
                                LOG('AutoFire','Hit Object But Not Player, Name='..self._auto_fire_detect_name)
                            end
                        end
                    end
                end
            end
        elseif self._autoFire_On then
            --延时检测阶段
            if not self.hasActiveAutoFire then
                if (_msec - self._autoDetect_timer) >= Config.Const.AutoFire_Config.AutoFire_ActiveTime then
                    self.hasActiveAutoFire = true
                    self._autoDetect_timer = _msec
                    self:pullTheTrigger(false, true)
                end
            else
                --停止判定
                if (_msec - self._autoDetect_timer) >= Config.Const.AutoFire_Config.AutoFire_Duration then
                    if not self.host.hasPlayerActiveTool then
                        local need_stop = self.weaponState == Const.WeaponState.DECOCKING
                        if not need_stop  then
                            local startPos = game.WorkSpace.CurrentCamera.Position
                            local dir = GameCamera.node.Rotation:LookDir()
                            local ret = WS:SweepSphere(10, startPos, dir, 2000, false, self.host.bulletCollisionGroup)
                            if (not ret) or (not ret.obj) or (not ret.obj:GetAttribute("HitPart")) then
                                need_stop = true 
                                LOG('AutoFire','#########停止射击')
                            else
                                self._autoDetect_timer = _msec
                            end
                        end

                        if need_stop then
                            self._autoFire_On = false
                            self.hasActiveAutoFire = false
                            self:releaseTheTrigger(true)
                        end
                    end
                end
            end
        end
    end,

    --@func 扣动扳机
    --@参数bSpecialAttack 是否是特殊攻击，比如刀得重击
    --@返回 无
    pullTheTrigger = function(self, special_attack, auto_fire)
        if not self.enable_control then return end
		if self.isLocalPlayer then
			if CMainGame.showCursor then
				return
			end
            --死亡状态，不能开枪
            if self.host.isDead then
                return
            end
		end

        local bSuc = false
        if self.weaponState == Const.WeaponState.READY then
			if self.isLocalPlayer then
                if not self:checkCapacityValid() then
					self:_setWeaponState(Const.WeaponState.DECOCKING)			--空仓挂机
					if self.isLocalPlayer then
						OUtils.playSound2D('fire_empty')
					end
					return
				end
			end
            if self:_isBow() and not auto_fire then
                if not auto_fire then
                    self.host:onHoldTrigger(true)
                else
                    self.host:onHoldTrigger(false)
                end
                self:_setWeaponState(Const.WeaponState.HOLDING)
            elseif self:_isGatlin() then
                self:_setWeaponState(Const.WeaponState.PRE_FIRE)
            else
                self:_setWeaponState(Const.WeaponState.FIRING)
            end
            bSuc = true
        elseif self.weaponState == Const.WeaponState.RELOADING then
            if tick() < self.fireStateTimer then
                return
            elseif self:checkCapacityValid() then
                --修改為打断上弹立即開火
                self:_setWeaponState(Const.WeaponState.FIRING)
                bSuc = true
            end
        elseif self.weaponState == Const.WeaponState.PRE_FIRE then
            if self.pre_fire_ready then
                self:_setWeaponState(Const.WeaponState.FIRING)
            end
            bSuc = true
        end

        self.special_attack = special_attack or false
        self.hasPulledTheTrigger = true
        self.hasReleasedTheTrigger = false

        if self.config.special_action == Const.WeaponSpecialAttack.FULL_AUTO then
            self:_fixFireRate(self.special_attack)
        end

        if not auto_fire then
            self._autoDetect_timer = tick()
            self._autoFire_On = false
            self.hasActiveAutoFire = false
        end

        if bSuc then
			if self.isLocalPlayer then
				--发送event
				Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_PULL,{weapon_id = self.cfg_id,attack_type=self.attack_type})
			end
        else
            --print('weapon pull the trigger State = '..self.weaponState..'  config_id = '..self.cfg_id)
        end
	    return bSuc
    end,

    --@func 释放扳机
    --@参数force 是否强制释放
    --@返回 无
    releaseTheTrigger = function(self,force)
        --trace_log('fire releaseTheTrigger')
        -- if not force then
        --     if self.weaponState == Const.WeaponState.FIRING and self:getSelectiveMode() == Const.ShootSelectiveMode.BURST then
        --         return
        --     end
        -- end
	    --松开扳机时，重置一下是否按下扳机
        self.hasPulledTheTrigger = false
        self.hasReleasedTheTrigger = true
        self.special_attack = false

        self._autoDetect_timer = tick()
        self._autoFire_On = false
        self.hasActiveAutoFire = false

        if self:_isBow() then
            self.host:onHoldTrigger(false)
        end

        if self:getSelectiveMode() ~= Const.ShootSelectiveMode.BURST then
            local seedMsg = false
            if (self.weaponState == Const.WeaponState.FIRING and self.firedRoundsPerPull >= 1) or not self.isLocalPlayer then
                -- if self.numOfBulletsInGun[self.mode] > 0 then
                if self:checkCapacityValid() then
                    self:_setWeaponState(Const.WeaponState.IDLE)
                else
                    self:_setWeaponState(Const.WeaponState.DECOCKING)
                end
                seedMsg = true
            elseif self.weaponState == Const.WeaponState.DECOCKING or self.weaponState == Const.WeaponState.IDLE then
                seedMsg = true
            end
    
            if self.weaponState == Const.WeaponState.HOLDING then
                self:_setWeaponState(Const.WeaponState.FIRING)
            end
    
            if self:_isGatlin() then
                self:_setWeaponState(Const.WeaponState.IDLE)
                self.host:onIdle()
                self.pre_fire_ready = false
            end
        end

		if seedMsg then
			if self.isLocalPlayer then
				--发送event
				Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_RELEASE,{weapon_id = self.cfg_id,ammo=self.numOfBulletsInGun})
			end
		end
        return true
    end,

    --@func 开关镜
    --@参数port 开or关
    --@参数force 强制
    --@返回 无
    portArms = function(self,port,force)
        if port then
            if not self.is_sight_on then
                if self.weaponState == Const.WeaponState.RELOADING or self.weaponState == Const.WeaponState.PREPARING then
                    return--上弹中不能开镜
                end
                self.is_sight_on = true
                self:_setWeaponState(Const.WeaponState.PORTING)
                if self.isInClient then
                    OUtils.playAnim('gun_port',self.player)
					--设置灵敏度
                    if self.isLocalPlayer then
					    GameCamera:setWeaponSensitivity(self.config.ADS_sensitivity)
                    end
					--设置FOV
                    if self.config.special_action == Const.WeaponSpecialAttack.SNIPER_AIM then
                        GameCamera:scaleFOV(self.config.ADS_FOV, -1)
                    else
                        GameCamera:scaleFOV(self.config.ADS_FOV, 0.3)
                    end
					--设置使用者的移速下降
					self.host:onGunPort(true,self.config.ADS_speed)

                    self.host:showAimUI(true, self.config.special_action == Const.WeaponSpecialAttack.SNIPER_AIM)
                end
            end
        else
            self.is_sight_on = false
            if not force then
                self:_setWeaponState(Const.WeaponState.READY)
            end
            if self.isInClient then
                --todo 怎么恢复到腰射状态
				--self.config.HIP_crossHair
				--设置灵敏度
                if self.isLocalPlayer then
                    GameCamera:setWeaponSensitivity(1)
                end
				--设置FOV
				-- GameCamera.node.FieldOfView = Const.DefaultFOV
                if self.config.special_action == Const.WeaponSpecialAttack.SNIPER_AIM then
                    GameCamera:scaleFOV(Const.DefaultFOV, -1)
                else
                    GameCamera:scaleFOV(Const.DefaultFOV, 0.3)
                end
                --OUtils.playAnim(self.player,'gun_port')
                if self.config.special_action == Const.WeaponSpecialAttack.SNIPER_AIM then
                    -- self.host:ShowHandle(true)
                    self.handle.Visible = true
                end

                --设置使用者的移速下降
				self.host:onGunPort(false, 1)

                self.host:showAimUI(false, self.config.special_action == Const.WeaponSpecialAttack.SNIPER_AIM)
            end
        end
    end,

    tryReload = function(self)
        LOG('RELOAD','GUN Config_id='..self.cfg_id)
        --死亡状态，不能开枪
        if not self:canReload() then
            LOG('RELOAD','ClientGun Cannot Reload capacityCarry='..self:getCapacityCarry()..' self:getMagazineCapacity()='..self:getMagazineCapacity()..' self:getCapacity()='..self:getCapacity()..'  ')
            return
        end
        if self.host.isDead then

        -- if self.numOfBulletsExtra[self.mode] <= 0 then

            if self.isLocalPlayer then
                self:_resetWeaponState(not self.is_sight_on)
            end
            LOG('RELOAD',' self:getCapacityCarry() <= 0')
        else
            self:reload()
        end
    end,

                else
                    return false
                end
            end
        else
            if self:_isOneByOneReloadType() then
                --装填
				self.host:onReload()
                self:_setWeaponState(Const.WeaponState.RELOADING)
            else
                --这里区别一下是否有弹膛留单，播放不同动作
                if self.bulletRemainingInBore then
                    OUtils.playAnim('gun_reload',self.player)
                else
                    --host_obj:event(_G.ModuleEvent.OnRoleWeaponState,'gun_reload',self.reloadTime)
                end
                --散弹枪不能加速换弹
                self.weaponStateTimer = tick() + self.cur_mode_data.weapon_reloadTime
                self.firstReload = false--true --改成false貌似时对的

                self:_setWeaponState(Const.WeaponState.RELOADING)
                self:_tryPlayReloadSound()
            end
            --散弹枪不能加速换弹
            --self.weaponStateTimer = world.msec + loadTime
            --self.firstReload = false--true --改成false貌似时对的
            --self:_setWeaponState(Const.WeaponState.RELOADING)
            return true
        end

        return false
    end,

    --@func 动画调用上弹逻辑可以执行了
    _animReload = function(self)
        if self:_isOneByOneReloadType() then
            --一发发装填
            local fill_full = true

            if self:getCapacity() < self:getMagazineCapacity() and self:getCapacityCarry() > 0 then--上满了或者没有余弹了
                --上一发
                -- self.numOfBulletsInGun[self.mode] = self.numOfBulletsInGun[self.mode] + 1
                self:setCapacity(self:getCapacity() + 1)
                self:setCapacityCarry(self:getCapacityCarry() - 1)
                if self.isLocalPlayer then
                    --todo 通知服务器
					Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_RELOAD,{weapon_id = self.cfg_id,ammo=self.numOfBulletsInGun,ammoExtra=self.numOfBulletsExtra})
                    self:_tryPlayReloadSound()
                    --通知当前持有武器数据变化 todo
					self.host:updateAmmo(self:getCapacity(), self:getCapacityCarry())
                    self:_setBulletsAttribute(true)
                end
                if self:getCapacity() < self:getMagazineCapacity() and self:getCapacityCarry() > 0 then --上满了或者没有余弹了
                    fill_full = false
                end
            end

            self.firedRoundsPerPull = 0

            if fill_full then
                self.host:setLocalPlayerActionState(Const.CharactorActionState.IDLE)
                self.firedRoundsPerPull = 0
                if self:_isManualBolt() and not self.haveFinishedCOCKING_WeaponIsManualBolt then
                    self:_setWeaponState(Const.WeaponState.COCKING)
                else
                    self:_setWeaponState(Const.WeaponState.READY)
                end
            end
        end
    end,

    --@func 更换弹匣
    _animReloadAll = function(self, _number)
        --引擎概率会多次调用关键帧，先写个逻辑代码规避一下
        if _number and self.numOfBulletsInGun == self:getMagazineCapacity() then
            _number = 0
        end
        --计算要添加的子弹数。
        local nAddNumOfBullets = _number or self:getMagazineCapacity() - self:getCapacity()--self.magazineCapacity[self.mode] - self.numOfBulletsInGun[self.mode]
        
        --更换弹夹式
        if self:getCapacityCarry() ~= -1 then
            nAddNumOfBullets = math.min(self:getCapacityCarry(), nAddNumOfBullets)
        end
    
        self:setCapacity(self:getCapacity() + nAddNumOfBullets)
        self:setCapacityCarry(self:getCapacityCarry() - nAddNumOfBullets)
        if self.isLocalPlayer then
            --todo 通知服务器
            Bridge:SendMessageToServer(Proto.ClientMSGID.CLIENT_WEAPON_RELOAD,{weapon_id = self.cfg_id,ammo=self.numOfBulletsInGun,ammoExtra=self.numOfBulletsExtra})
            --todo 通知当前持有武器数据变化
            self.host:updateAmmo(self:getCapacity(), self:getCapacityCarry())
            self:_setBulletsAttribute(true)
        end
    
        self.firedRoundsPerPull = 0
        if self:_isManualBolt() and not self.haveFinishedCOCKING_WeaponIsManualBolt then
            self:_setWeaponState(Const.WeaponState.COCKING)
        else
            self:_setWeaponState(Const.WeaponState.READY)
        end
    end,

    --@func 更新精度
    --@参数
    --@返回 无
    _updateAccuracy = function(self,afterShot)
        local index = self:_getStateIndex()

        --这里使用调节过的子弹发射数来计算精度，因为想达到在切换弹夹后，精度达到最大值
        local nAdjustShotsFired = false--int(m_fShotsFiredAdjustParaForAccuracy);
        if afterShot then
            nAdjustShotsFired = true
            --[[if self.shotsFired > 0 then
                nAdjustShotsFired = true
            elseif self.valueChangeForAccuracy_x.changing then--准星不在扩散，第一枪不改变扩散
                nAdjustShotsFired = true
            end]]
        end
        if nAdjustShotsFired then
            self.accuracy_x = math.min(self.cur_mode_data.spread_max[index], self.valueChangeForAccuracy_x.cur + self.cur_mode_data.spread_radius_correct[index])
            self.accuracy_y = 0.5*OUtils.toRadian(self.valueChangeForAccuracy_y.cur+self.cur_mode_data.spread_range_correct[index])
        else
            self.accuracy_x = math.min(self.cur_mode_data.spread_max[index], self.cur_mode_data.spread_radius[index])
            self.accuracy_y = 0.5*OUtils.toRadian(self.cur_mode_data.spread_range[index])
        end

        --精度平滑改变
        if true then
            local accSmooth = function(valueChangeForAccuracy,accuracy)
                local oldAccuracy = valueChangeForAccuracy.dest
                if oldAccuracy ~= accuracy then --精度发生了改变
                    if self.weaponState == Const.WeaponState.FIRING or self.weaponState == Const.WeaponState.FIRING_EX then
                        if accuracy > oldAccuracy then --开枪中增加则突变
                            local changeTime = 0
                            valueChangeForAccuracy:stop()
                            valueChangeForAccuracy:start(oldAccuracy, accuracy, changeTime)
                        else
							local changeTime = math.abs(accuracy - valueChangeForAccuracy.cur)*self.cur_mode_data.spread_cooling_time*0.001
                            valueChangeForAccuracy:stop()
                            valueChangeForAccuracy:start(valueChangeForAccuracy.cur, accuracy, changeTime)
                        end
                    else
						local changeTime = math.abs(accuracy - valueChangeForAccuracy.cur)*self.cur_mode_data.spread_cooling_time*0.001
                        valueChangeForAccuracy:stop()
                        valueChangeForAccuracy:start(valueChangeForAccuracy.cur, accuracy, changeTime)
                    end
                end
            end
            accSmooth(self.valueChangeForAccuracy_x,self.accuracy_x)
            accSmooth(self.valueChangeForAccuracy_y,self.accuracy_y)
        end
    end,

    --@func 是否需要一颗一颗子弹的换弹，比如霰弹枪
    --@参数
    --@返回 无
    _isOneByOneReloadType = function(self)
        if self.cur_mode_data.reload_type == 3 then
            local _canAdd = self.magazineCapacity - self.numOfBulletsInGun  
            if self.numOfBulletsExtra < _canAdd then
                _canAdd = self.numOfBulletsExtra
            end
            if (_canAdd/5) >= 1 then
                return false
            else
                return true
            end
        end
        return self.cur_mode_data.reload_type == 2
    end,

    --获得一次性上弹的数量
    getReloadSubType = function(self)
        local _canAdd = self.magazineCapacity - self.numOfBulletsInGun  
        if self.numOfBulletsExtra < _canAdd then
            _canAdd = self.numOfBulletsExtra
        end
        if _canAdd >= 10 then
            return 2
        elseif _canAdd >= 5 then
            return 1
        else
            return 0
        end
    end,

    --两种上弹方式混合的枪械，计算一下第一次上弹结束剩余多少颗子弹没有上
    getReloadRemains = function(self)
        local reloadRemains = 0
        local reloadRemains = self.magazineCapacity - self.numOfBulletsInGun  
        if self.numOfBulletsExtra < reloadRemains then
            reloadRemains = self.numOfBulletsExtra
        end
        while reloadRemains >= 5 do
            reloadRemains = reloadRemains - 5
        end

        return reloadRemains
    end,

    --动画事件回调
    onAnimCallBack = function(self,action, state)
        if state == Enum.StateMachineMessage.kOnStateExit then
            if action == Const.CharactorActionState.POST_RELOAD then
            elseif action == Const.CharactorActionState.LOAD then
                self:loadEnd()
            elseif action == Const.CharactorActionState.FIRE then
                if self:_isBow() then
                    self:_animReload() --弓箭直接在动画结束时上弹
                elseif not self:checkCapacityValid() then
                    self:_setWeaponState(Const.WeaponState.DECOCKING)
                end
            elseif action == Const.CharactorActionState.CHANGING
            or action == Const.CharactorActionState.START then
                if self.isLocalPlayer then
                    if not self:checkCapacityValid() then
                        self:_setWeaponState(Const.WeaponState.DECOCKING)
                    elseif self:_isManualBolt() and not self.haveFinishedCOCKING_WeaponIsManualBolt then
                        self.host:onLoad()
                        self:_setWeaponState(Const.WeaponState.COCKING_PREPARE)
                    else
                        self:_setWeaponState(Const.WeaponState.READY)
                    end
                    self.host:updateAmmo(self:getCapacity(), self:getCapacityCarry())
                end
            end
        elseif state == Enum.StateMachineMessage.kOnStateEnter then
            if action == Const.CharactorActionState.CHANGING then
                if self.config.special_action == Const.WeaponSpecialAttack.M_AIM then
                    --开镜操作的结果
                    if self.host.gun_mode == 2 then
                        self:portArms(true)
                        self:_portDone()
                    else
                        self:portArms(false)
                    end
                end
            elseif action == Const.CharactorActionState.FIRE_POST_HOLD then
                self:_fireBullet()
            elseif action == Const.CharactorActionState.RELOAD then
                self:_animReload()
            end
        end
        
        if self.bullet_node then
            self.bullet_node.Visible = true
        end
    end,

    onAnimClipEvent = function(self, key, value)
        if key == 'RELOAD_TIME' then
            if value == 'SKS' then
                local number = 5
                local reload_subtype = self:getReloadSubType()
                if reload_subtype == 2 then
                    number = 10
                end
                self:_animReloadAll(number)
            else
                self:_animReloadAll()
            end
        elseif key == 'HideBullet' then
            if self.bullet_node then
                self.bullet_node.Visible = false
            end
        elseif key == 'ShowBullet' then
            if self.bullet_node then
                self.bullet_node.Visible = true
            end
        elseif key == 'SPECIAL_TIME' then
            --2226右键操作动画结束.播放特效
            EffectManager:playWeaponEffect(EffectDefine.EnumEffectType.eWeaponPart, {weaponId = self.cfg_id, parent = self.Muzzle, charactorFps = self.handle.Parent.Parent, isRightKeyOption = true})
            self:playEntityBulletEffect()
        elseif key == 'fire_bullet' then
            self:_fireBullet()
        elseif key == 'run_script' then
            local script = require(WeaponSystem:WaitForChild(value))
            script:fire(self)
        end
    end,
