local MainStorage   = game:GetService("MainStorage")
local Class         = require(MainStorage.Scripts.Class)
local BaseFireShake = Class()

--[[
  三轴震动配置说明:
  pitch (X轴抬头): 开枪后的抬头抖动
  yaw   (Y轴摆头): 左右随机摆动
  roll  (Z轴旋转): 枪身旋转抖动

  每轴参数:
  amplitude  : 最大振幅(角度)，越大震动越猛
  decayRate  : 衰减率，越大恢复越快
  frequency  : 震动频率(Hz)，越大抖动越碎
  duration   : 持续时间(秒)，控制这一轴震多久

]]

function BaseFireShake:ctor()
    self.elapsed = 0
    self.isFiring = false
    self.pitchRand = 0
    self.yawRand = 0
    self.rollRand = 0
end

function BaseFireShake:Init(fireShake)
    local cfg = fireShake or {}

    self.pitch = {
        amplitude = cfg.pitchAmplitude or 3,
        decayRate = cfg.pitchDecayRate or 8,
        frequency = cfg.pitchFrequency or 12,
        duration  = cfg.pitchDuration or 0.15,
    }
    self.yaw = {
        amplitude = cfg.yawAmplitude or 1.5,
        decayRate = cfg.yawDecayRate or 6,
        frequency = cfg.yawFrequency or 10,
        duration  = cfg.yawDuration or 0.2,
    }
    self.roll = {
        amplitude = cfg.rollAmplitude or 0.8,
        decayRate = cfg.rollDecayRate or 10,
        frequency = cfg.rollFrequency or 15,
        duration  = cfg.rollDuration or 0.12,
    }
end

function BaseFireShake:OnFire()
    self.elapsed = 0
    self.isFiring = true

    -- 每次开枪生成随机种子，让每一枪的抖动都不同
    self.pitchRand = math.random() * 0.3 - 0.15
    self.yawRand = math.random() * 0.5 - 0.25
    self.rollRand = math.random() * 0.4 - 0.2
end

function BaseFireShake:Update(dt)
    if not self.isFiring then
        return
    end

    self.elapsed = self.elapsed + dt
    local t = self.elapsed

    local function calcAxis(axis)
        if t > axis.duration then
            return 0
        end
        local decay = math.exp(-axis.decayRate * t)
        local wave = math.sin(2 * math.pi * axis.frequency * t)
        return axis.amplitude * decay * wave * math.rad(1)
    end

    local function calcAxisPitch(axis)
        if t > axis.duration then
            return 0
        end
        local decay = math.exp(-axis.decayRate * t)
        local wave = math.sin(2 * math.pi * axis.frequency * t + self.pitchRand)
        return axis.amplitude * decay * wave * math.rad(1)
    end

    local pitchOffset = calcAxisPitch(self.pitch)
    local yawOffset   = calcAxis(self.yaw)   * (1 + self.yawRand)
    local rollOffset  = calcAxis(self.roll)  * (1 + self.rollRand)

    if t > math.max(self.pitch.duration, self.yaw.duration, self.roll.duration) then
        self.isFiring = false
    end

    return Vector3.New(pitchOffset, yawOffset, rollOffset)
end

function BaseFireShake:IsFinished()
    return not self.isFiring
end
