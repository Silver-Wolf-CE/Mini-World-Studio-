local blaster_map = {}
local clsList_map = {}
local default_sound_config = {
    soundType = "Blaster",
    soundFolder = "AWP",
    switchSound = "SwitchGun",
    shootSound = "AWPShoot",
    reloadSound = "AWPReload",
    boltActionSound = "AWPBoltAction",
}

local function init_blaster_config()
	for index, value in ipairs(blaster_config) do
		for soundKey, soundValue in pairs(default_sound_config) do
			if value[soundKey] == nil then
				value[soundKey] = soundValue
			end
		end
		blaster_map[value.id] = value
		if not clsList_map[value.clsType] then
			clsList_map[value.clsType] = {}
		end
		table.insert(clsList_map[value.clsType] , value)
	end
end

init_blaster_config()

local super = require(script.Parent)
local BlasterConfig = require(rs.Scripts.Class)(super)

function BlasterConfig:ctor()

end

function BlasterConfig:Init()

end

function BlasterConfig:GetBlasterConfig()
	return blaster_config
end

function BlasterConfig:GetBlasterConfigById(id)
	return blaster_map[id or 0]
end

--根据clsType获取对应的BlasterConfig
function BlasterConfig:GetBlasterConfigByClsType(clsType)
    return clsList_map[clsType] or {}
end

return BlasterConfig
