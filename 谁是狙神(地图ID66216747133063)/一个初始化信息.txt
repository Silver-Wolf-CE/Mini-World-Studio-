local function GetChild(parent, childName)
	if not parent or not childName then
		return nil
	end
	if parent.FindFirstChild then
		return parent:FindFirstChild(childName)
	end
	return parent[childName]
end

local function ReadSkinId(skinNode)
	if not skinNode then
		return nil
	end
	local ok, skinId = pcall(function()
		return skinNode.SkinId
	end)
	skinId = ok and tonumber(skinId) or nil
	if skinId and skinId > 0 then
		return skinId
	end
	return nil
end

function Player:Init()
	self.entityId = localPlayer.UserId
	-- 事件管理类优先初始化
	self.EventManager = GlobalEventManager

	for index, value in pairs(PlayerModule) do
		local model = PlayerData:FindFirstChild(value .. "Data", 5)
		if model then
			local m = require(model)
			if m and m.New then
				self[value] = m.New(self)
			else
				shared.warn("player module load error:" .. value)
			end
		else
			shared.warn("player module load error:" .. value)
		end
	end
	for index, value in pairs(PlayerModule) do
		local m = self[value]
		if m then
			m:Init()
		end
	end
end

-- 模拟数据
function Player:ApplyLocalSkinId(skinId)
	local character = localPlayer and localPlayer.Character
	if character and skinId then
		character.SkinId = skinId
	end
end

function Player:SelectRoleSkinId(skinId)
	skinId = tonumber(skinId)
	if not skinId or skinId <= 0 then
		return
	end
	self:ApplyLocalSkinId(skinId)
	NetMessage:SendRequest(ProtoId.Player_ChangeSkinReq, {skinId = skinId})
end

function Player:BindSelectRoleTrigger(root, roleConfig)
	local skinNode = GetChild(root, roleConfig.skinNodeName)
	local triggerBox = GetChild(root, roleConfig.triggerName)
	local skinId = ReadSkinId(skinNode)
	if not triggerBox then
		shared.warn("[Player] missing select role trigger: " .. tostring(roleConfig.triggerName))
		return
	end
	if not skinId then
		shared.warn("[Player] missing select role SkinId: " .. tostring(roleConfig.skinNodeName))
		return
	end
	table.insert(self.selectRoleConnList, triggerBox.Touched:Connect(function(node)
		if localPlayer and localPlayer.Character and node == localPlayer.Character then
			self:SelectRoleSkinId(skinId)
		end
	end))
end

function Player:InitSelectRoleTrans()
	if self.isSelectRoleInit then
		return
	end
	self.isSelectRoleInit = true
	local root = GetChild(game:GetService("WorkSpace"), "SelectRoleTrans")
	if not root then
		shared.warn("[Player] missing Workspace.SelectRoleTrans")
		return
	end
	for _, roleConfig in ipairs(SELECT_ROLE_CONFIG) do
		self:BindSelectRoleTrigger(root, roleConfig)
	end
end

-- 获取道具数量
function Player:GetItemNum(id)
	local config = ConfigManager.ItemConfig:GetItemConfigById(id)
	if not config then return 0 end
	if config.type == ItemEnum.ItemType.currency then
		-- 货币
		return self.Detail:GetCurrencyNum(id)
	else
		-- 获取道具
		return self.Bag:GetItemNumByConfigId(id)
	end
end


-- 获取道具数量
function Player:CheckItemEnough(items, notify)
	if items == nil or #items == 0 then
		return false	-- 如果为空/空表返回false, 防止配置错误被刷物品,不需要消耗物品的话,就无需调此接口
	end
	for index, value in ipairs(items) do
		if not value.id or not value.num then	-- 结构不对
			return false
		end
		local have = self:GetItemNum(value.id)
		if have < value.num then
			if notify then
				local config = ConfigManager.ItemConfig:GetItemConfigById(value.id)
				if config then
					self:SendTips(string.format(TextStr.NotEnoughTipsStr, config.name, shared.NumToString(value.num), shared.NumToString(have)))
				end
			end
			return false
		end
	end
	return true
end

--检查 不弹商店购买
function Player:CheckItemEnough2(items, notify , isGuide)
	if items == nil or next(items) == nil then
		return true
	end
	for index, value in ipairs(items) do
		local have = self:GetItemNum(value.id)
		if have < value.num then
			if notify then
				local config = ConfigManager.ItemConfig:GetItemConfigById(value.id)
				if config then	
					self:SendTips(string.format(TextStr.NotEnoughTipsStr, config.name, shared.NumToString(value.num), shared.NumToString(have)))
				end
			end
			return false
		end
	end
	return true
end
