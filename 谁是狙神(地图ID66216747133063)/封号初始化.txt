function ChatManager:GMCommand(cmd, ...)
	local data = {
		cmd = cmd,
		params = {...}
	}

	NetMessage:SendRequest(ProtoId.GM_GMCommand, data)
end

function ChatManager:OpenPlayerUI(cmd, ...)
	local params = {...}
	local uiName = params[1]
	if not uiName then
		return
	end
	local player = require(rs.Scripts.Entity.Player)
	player.UIManager:OpenUI(uiName)
end

function ChatManager:OpenEvolutionSuccUI(cmd, ...)
	local params = {...}
	local animalId = tonumber(params[1])
	if not animalId then
		return
	end
	local player = require(rs.Scripts.Entity.Player)
	player.UIManager:OpenUI("EvolutionSucc", animalId)
end
